import { firebaseConfig } from "./js/firebase-config.js"; const app = initializeApp(firebaseConfig); const db = getFirestore(app); // ── Estado global ── let todosLosPedidos = []; let filtroActivo = "todos"; if (!lista.length) { c.innerHTML = `

Sin pedidos con ese filtro.

`; return; } c.innerHTML = lista.map(p => { const ini = iniciales(p.cliente); const dot = dotClass(p.estado); const esDom = (p.metodoEntrega || "").toLowerCase().includes("domicilio"); const productosHTML = Array.isArray(p.productos) ? p.productos.map(pr => `
  • ${pr.cantidad}× ${pr.nombre}$${(pr.precio * pr.cantidad).toFixed(2)}
    • `).join("") : "
  • Sin detalle
    • "; const cuponTag = p.cuponUsado && p.cuponUsado !== "Ninguno" ? `🎟 ${p.cuponUsado}` : ""; return `

    ${ini}
    ${p.cliente || "Cliente"}

      ${productosHTML}
    ${p.metodoEntrega || "Recoger en tienda"}

    ` : ""}

    `; }).join(""); } // ── Filtrar + buscar ── function aplicarFiltros() { let lista = [...todosLosPedidos]; if (filtroActivo !== "todos") lista = lista.filter(p => (p.estado || "").includes(filtroActivo)); if (busquedaActual) { const q = busquedaActual.toLowerCase(); lista = lista.filter(p => (p.cliente || "").toLowerCase().includes(q) || (p.codigoPedido || "").toLowerCase().includes(q) ); } renderPedidos(lista); } window.filtrarPedidos = (filtro, btn) => { filtroActivo = filtro; document.querySelectorAll(".filtro-chip").forEach(b => b.classList.remove("active")); btn.classList.add("active"); aplicarFiltros(); }; window.buscarEnPedidos = (val) => { busquedaActual = val; aplicarFiltros(); }; // ── Cambiar estado ── window._cambiarEstado = async (id, nuevoEstado) => { try { await updateDoc(doc(db, "pedidos", id), { estado: nuevoEstado }); const p = todosLosPedidos.find(x => x._id === id); if (p) p.estado = nuevoEstado; actualizarKPIs(todosLosPedidos); // Actualizar solo el dot sin re-renderizar todo const card = document.getElementById("card-" + id); if (card) { const dot = card.querySelector(".pedido-estado-dot"); dot.className = "pedido-estado-dot " + dotClass(nuevoEstado); } Swal.fire({ toast: true, position: "top-end", icon: "success", title: "Estado actualizado", showConfirmButton: false, timer: 1600 }); } catch (e) { Swal.fire("Error", "No se pudo actualizar el estado.", "error"); } }; // ── Cargar pedidos ── async function cargarPedidos() { document.getElementById("contenedorPedidosAdmin").innerHTML = `
